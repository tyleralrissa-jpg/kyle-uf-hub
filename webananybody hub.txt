    local function configFileName(name)
        name = tostring(name or ""):gsub("[^%w_%-%s]", ""):gsub("%s+", "_")
        if name == "" then name = "default" end
        return CONFIG_FOLDER .. "/" .. name .. ".json", name
    end

    local function canUseFiles()
        return type(writefile) == "function" and type(readfile) == "function"
            and type(isfile) == "function" and type(makefolder) == "function"
            and type(isfolder) == "function"
    end

    local function ensureConfigFolder()
        if not canUseFiles() then return false end
        if not isfolder(CONFIG_FOLDER) then
            pcall(makefolder, CONFIG_FOLDER)
        end
        return isfolder(CONFIG_FOLDER)
    end

    local function listSavedConfigs()
        if not ensureConfigFolder() then return {} end
        local names = {}
        local ok, files = pcall(function()
            if type(listfiles) == "function" then
                return listfiles(CONFIG_FOLDER)
            elseif type(listfolder) == "function" then
                return listfolder(CONFIG_FOLDER)
            end
            return nil
        end)
        if ok and type(files) == "table" then
            for _, path in ipairs(files) do
                local pathStr = tostring(path)
                if pathStr:lower():match("%.json$") then
                    local name = pathStr:match("([^/\\]+)%.json$") or pathStr:match("([^/\\]+)$")
                    if name and name ~= "" and name ~= "last" then
                        table.insert(names, name)
                    end
                end
            end
        end
        table.sort(names, function(a, b) return a:lower() < b:lower() end)
        return names
    end

    local function deleteConfig(name)
        if not ensureConfigFolder() then return false, "File system not supported." end
        local path = select(1, configFileName(name))
        if not isfile(path) then return false, "Config not found." end
        if type(delfile) == "function" then
            local ok = pcall(delfile, path)
            if ok then return true, "Deleted: " .. tostring(name) end
            return false, "Could not delete config."
        end
        return false, "Delete not supported by this executor."
    end

    -- ── BALL DETECTION LOGIC ──────────────────────────────────────────
    local function isValidBall(part)
        if not part or not part:IsA("BasePart") then return false end
        if not part:IsDescendantOf(workspace) then
            local char = plr.Character
            if char and part:IsDescendantOf(char) then
                local lowerName = part.Name:lower()
                if lowerName:find("ball") or lowerName:find("football") then
                    return true
                end
            end
            return false
        end
        return true
    end

    local function findActiveBall()
        local char = plr.Character
        if char then
            for _, desc in ipairs(char:GetDescendants()) do
                if desc:IsA("BasePart") then
                    local lowerName = desc.Name:lower()
                    if lowerName:find("ball") or lowerName:find("football") then
                        return desc
                    end
                end
            end
        end

        local tagged = CollectionService:GetTagged(TAG_NAME)
        for _, obj in ipairs(tagged) do
            if isValidBall(obj) then return obj end
        end
        
        local found = workspace:FindFirstChild(BALL_NAME, true)
        if isValidBall(found) then return found end

        for _, folderName in pairs({"Games", "MiniGames", "Live", "Balls", "Workspace", "Active", "Field"}) do
            local folder = workspace:FindFirstChild(folderName)
            if folder then
                local folderBall = folder:FindFirstChild(BALL_NAME, true)
                if isValidBall(folderBall) then return folderBall end
            end
        end

        for _, desc in ipairs(workspace:GetDescendants()) do
            if desc:IsA("BasePart") and (desc.Name:lower():find("ball") or desc.Name:lower():find("football")) then
                if isValidBall(desc) then return desc end
            end
        end
        
        return nil
    end

    CollectionService:GetInstanceAddedSignal(TAG_NAME):Connect(function(obj)
        if isValidBall(obj) then cachedBall = obj end
    end)

    workspace.ChildAdded:Connect(function(child)
        if (child.Name == BALL_NAME or child.Name:lower():find("ball")) and isValidBall(child) then

    local function applyTackleReach()
        if not TR_ENABLED then return end
        local char = plr.Character
        if not char then return end

        for _, folderName in pairs({"Games", "MiniGames", "ParkMap", "ParkMatchMap"}) do
            local folder = Workspace:FindFirstChild(folderName)
            if folder then
                for _, instance in pairs(folder:GetChildren()) do
                    local rep = instance:FindFirstChild("Replicated")
                    if rep and rep:FindFirstChild("Hitboxes") then
                        for _, part in pairs(rep.Hitboxes:GetChildren()) do
                            if part:IsA("BasePart") and (part.Name == plr.Name or part.Name == tostring(plr.UserId)) then
                                if part.Size ~= Vector3.new(TR_SIZE, TR_SIZE * 1.5, TR_SIZE) then
                                    part.Size = Vector3.new(TR_SIZE, TR_SIZE * 1.5, TR_SIZE)
                                    part.Transparency = TR_TRANS
                                    part.Material = Enum.Material.Neon
                                    part.Color = Color3.fromRGB(220, 35, 35)
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    -- ── STICKY HEAD HELPER FUNCTIONS ──────────────────────────────────
    local function getClosestPlayer()
        local nearest = nil
        local minDist = 35
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= plr and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChild("Humanoid")
                if hrp and hum and hum.Health > 0 and root then
                    local d = (hrp.Position - root.Position).Magnitude
                    if d < minDist then
                        minDist = d
                        nearest = p
                    end
                end
            end
        end
        return nearest
    end

    local function doHeadBoost(rootPart)
        onBoostCooldown = true
        boostCdRemaining = BOOST_COOLDOWN

        local bv = Instance.new("BodyVelocity")
        bv.Velocity = Vector3.new(0, BOOST_POWER, 0)
        bv.MaxForce = Vector3.new(0, 9e9, 0)
        bv.Parent = rootPart
        task.wait(0.2)
        if bv then bv:Destroy() end

        task.delay(BOOST_COOLDOWN, function()
            onBoostCooldown = false
        end)
    end

    local function checkFeetOnHead(myHRP, theirHead)
        local myFeetY = myHRP.Position.Y - (myHRP.Size.Y / 2)
        local headTopY = theirHead.Position.Y + (theirHead.Size.Y / 2)

        local dy = myFeetY - headTopY
        if dy < -0.5 or dy > 2 then return false end

        local dx = math.abs(myHRP.Position.X - theirHead.Position.X)
        local dz = math.abs(myHRP.Position.Z - theirHead.Position.Z)
        local halfX = theirHead.Size.X / 2 + 0.5
        local halfZ = theirHead.Size.Z / 2 + 0.5

        return dx <= halfX and dz <= halfZ
    end

        if not character or not root then return end

        if JUMP_SPEED_ENABLED and JUMP_MULTIPLIER > 1.0 and humanoid and humanoid:GetState() ~= Enum.HumanoidStateType.Freefall then
            local speedBoost = (JUMP_MULTIPLIER - 1) * 18

                root.AssemblyLinearVelocity.Z

            jumpOnCD = true
            root:ApplyImpulse(Vector3.new(0, JUMP_POWER_MULTIPLIER * root.AssemblyMass, 0))
            task.wait(JUMP_COOLDOWN)
            jumpOnCD = false
        end
    end)

            local percentage = relativeX / trackSize
            local val        = minVal + (percentage * (maxVal - minVal))

            if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then

            end
        end)

        end

    RunService.RenderStepped:Connect(function(dt)
        local now = os.clock()

        if TR_ENABLED and (now - lastReachUpdate > 0.1) then
            lastReachUpdate = now
            applyTackleReach()
        end

        if HITBOX_ENABLED and (now - lastHitboxUpdate > 0.1) then
            lastHitboxUpdate = now
            local targetSize = Vector3.new(HITBOX_SIZE, HITBOX_SIZE, HITBOX_SIZE)
            for _, v in ipairs(Players:GetPlayers()) do
                if v ~= plr then
                    local hrp = v.Character and v.Character:FindFirstChild("HumanoidRootPart")
                    if hrp and hrp.Size ~= targetSize then
                        hrp.Size = targetSize
                        hrp.Transparency = 1
                        hrp.Material = Enum.Material.Neon
                        hrp.CanCollide = true
                    end
                end
            end
        end

        if not cachedBall or not cachedBall.Parent or not cachedBall:IsDescendantOf(workspace) then
            local found = findActiveBall()
            if found then cachedBall = found end
        end

        if not cachedBall or not isValidBall(cachedBall) then
            if now - lastBallSearch > 0.3 then
                lastBallSearch = now
                cachedBall = findActiveBall()
            end
        end

        local currentDist = nil
        if root and cachedBall and cachedBall.Parent then
            local char = plr.Character
            if char and cachedBall:IsDescendantOf(char) then
                currentDist = 0.0
            else
                currentDist = (cachedBall.Position - root.Position).Magnitude
            end
            distanceLabel.Text = "BALL DISTANCE\n" .. string.format("%.1f", currentDist) .. " studs"
        else
            distanceLabel.Text = "BALL DISTANCE\nSearching..."
        end

        if isAirborne then
            stateLabel.Text = "STATE\nAirborne"
            stateLabel.TextColor3 = COL_ON
        else
            stateLabel.Text = "STATE\nGrounded"
            stateLabel.TextColor3 = COL_DIM
        end

        if PROX_GRAVITY_ENABLED then
            local char = plr.Character
            local holdingBall = char and cachedBall and cachedBall:IsDescendantOf(char)
            local shouldBeLowGravity = holdingBall or lockedInAir or (currentDist and currentDist <= TRIGGER_RADIUS)

            if shouldBeLowGravity then
                if Workspace.Gravity ~= LOW_GRAVITY then Workspace.Gravity = LOW_GRAVITY end
                gravityValue.Text = string.format("%.1f", LOW_GRAVITY)
                gravityValue.TextColor3 = COL_ON
            else
                if Workspace.Gravity ~= DEFAULT_GRAVITY then Workspace.Gravity = DEFAULT_GRAVITY end
                gravityValue.Text = string.format("%.1f", DEFAULT_GRAVITY)
                gravityValue.TextColor3 = COL_ACCENT
            end
        end

        if humanoid and root then
            local velY = root.AssemblyLinearVelocity.Y
            if velY < -2 then
                workspace.Gravity = SUPER_FALL_GRAVITY
            elseif not PROX_GRAVITY_ENABLED then
                workspace.Gravity = NORMAL_GRAVITY
            end
        end

        if STICKY_HEAD_ENABLED and humanoid and root then
            local target = getClosestPlayer()
            if target and target.Character then
                local targetHead = target.Character:FindFirstChild("Head")
                if targetHead and humanoid.FloorMaterial == Enum.Material.Air then
                    local headPos = targetHead.Position + Vector3.new(0, 1.6, 0)
                    local offset = headPos - root.Position
                    local dist = offset.Magnitude
                    local vel = root.AssemblyLinearVelocity

                    if dist > 2.5 then
                        local multiplier = STICKY_PULL * 18
                        local desired = offset.Unit * multiplier
                        root.AssemblyLinearVelocity = Vector3.new(
                            vel.X + (desired.X - vel.X) * 0.25,
                            vel.Y,
                            vel.Z + (desired.Z - vel.Z) * 0.25
                        )
                    end

                    if dist <= 3 then
                        local centerDir = (headPos - root.Position)
                        local horizontal = Vector3.new(centerDir.X, 0, centerDir.Z)
                        root.AssemblyLinearVelocity += horizontal.Unit * (STICKY_POWER * 5)
                    end

                    if dist <= 2 then
                        local boostOffset = headPos - root.Position
                        root.AssemblyLinearVelocity = Vector3.new(
                            boostOffset.X * (STICKY_POWER * 5),
                            root.AssemblyLinearVelocity.Y,
                            boostOffset.Z * (STICKY_POWER * 5)
                        )
                    end

                    if dist <= 1.2 then
                        local lock = headPos - root.Position
                        root.AssemblyLinearVelocity = Vector3.new(
                            lock.X * (STICKY_POWER * 6),
                            root.AssemblyLinearVelocity.Y,
                            lock.Z * (STICKY_POWER * 6)
                        )
                    end
                end
            end
        end
    end)

    RunService.Heartbeat:Connect(function(dt)
        if boostCdRemaining > 0 then
            boostCdRemaining = math.max(0, boostCdRemaining - dt)
            boostStateLabel.Text = string.format("HEAD BOOST CD: %.1fs", boostCdRemaining)
            boostStateLabel.TextColor3 = COL_WARN
        else
            boostStateLabel.Text = "HEAD BOOST: READY"
            boostStateLabel.TextColor3 = COL_SUCCESS
        end

        if not STICKY_HEAD_ENABLED or onBoostCooldown then return end

        local myChar = plr.Character
        if not myChar then return end
        local myHRP = myChar:FindFirstChild("HumanoidRootPart")
        if not myHRP then return end

        for _, p in ipairs(Players:GetPlayers()) do
            if p == plr then continue end
            local char = p.Character
            if not char then continue end

            local hum = char:FindFirstChildOfClass("Humanoid")
            local theirHead = char:FindFirstChild("Head")
            if not hum or not theirHead then continue end

            local wasJumping = playerJumpStates[p] or false
            local isJumping = hum.FloorMaterial == Enum.Material.Air

            if isJumping and not wasJumping then
                playerJumpTimers[p] = 0.15
            end

            if playerJumpTimers[p] and playerJumpTimers[p] > 0 then
                playerJumpTimers[p] = playerJumpTimers[p] - dt
                if checkFeetOnHead(myHRP, theirHead) then
                    playerJumpTimers[p] = 0
                    doHeadBoost(myHRP)
                end
            end

            playerJumpStates[p] = isJumping
        end
    end)

    Players.PlayerRemoving:Connect(function(p)
        playerJumpStates[p] = nil
        playerJumpTimers[p] = nil
    end)

    UserInputService.InputBegan:Connect(function(input, isProcessed)
        if isProcessed then return end
        
        local key = input.KeyCode
        
        if key == Enum.KeyCode.V or key == Enum.KeyCode.ButtonL1 then
            WALK_SPEED_ENABLED = not WALK_SPEED_ENABLED
            if humanoid then
                humanoid.WalkSpeed = WALK_SPEED_ENABLED and CUSTOM_WALKSPEED or DEFAULT_WALKSPEED
            end
            if refreshSpeedBtn then refreshSpeedBtn() end
        end
        
        if key == Enum.KeyCode.R or key == Enum.KeyCode.ButtonR1 then
            STICKY_HEAD_ENABLED = not STICKY_HEAD_ENABLED
            if refreshStickyBtn then refreshStickyBtn() end

            if RANDOM_BOOST_ENABLED then
                BOOST_POWER = math.random(21, 70)
                for _, refresh in ipairs(configRefreshers) do
                    pcall(refresh)
                end
            end
        end

        if key == TOGGLE_KEY or key == Enum.KeyCode.H or key == Enum.KeyCode.LeftControl or key == Enum.KeyCode.ButtonSelect or key == Enum.KeyCode.DPadRight then
            if panel then
                panel.Visible = not panel.Visible
                shadow.Visible = panel.Visible
            end
        end
    end)
end
